Standard Anthropic Messages API (Safest for Text-Only Outputs)

      anthropic Python SDK package


<<<python
import anthropic

client = anthropic.Anthropic()

    # Pass system requirements and formatting rules directly to the model
response = client.messages.create(
    model="claude-3-5-sonnet-20240620",  # Or your preferred frontier model
    max_tokens=2048,
    system="You are an expert Python engineer. Return ONLY valid Python code inside markdown blocks. Do not explain the code, do not use tools, and provide instructions for dependencies inside a comment block at the top of the file.",
    messages=[
        {
            "role": "user",
            "content": "System Requirements: Mac M-series, Python 3.10+, ffmpeg installed. Prompt: Generate a python script to translate audio files from Spanish to English using Faster-Whisper."
        }
    ]
)

print(response.content[0].text)

Option A: Using the Claude Agent SDK (With a Custom System Prompt)
Anthropic allows you to modify the system prompt of the Agent SDK. This strips the model of its agentic behavior and enforces standard code generation

<<<python
import asyncio
from claude_agent_sdk import query, ClaudeAgentOptions

async def generate_pure_code():
    # Enforce strict system instructions to disable tool-use expectations
    custom_system_prompt = """
    You are a strict Python code generation assistant. 
    Your ONLY purpose is to return cleanly formatted Python code text. 
    - DO NOT assume you have access to tool-use, a bash terminal, or file operations.
    - Provide all setup commands purely as text comments within the code block.
    - Output raw markdown text for copy-pasting.
    """

    prompt = """
    Create a Python speech-to-text translation script.
    System Requirements: Python 3.11, PyAudio, OpenAI-Whisper.
    Target: Transcribe live microphone audio to English text.
    """

    # Query the SDK with the overridden system prompt
    async for message in query(prompt=prompt, options={"systemPrompt": custom_system_prompt}):
        if message.type == "assistant":
            print(message.message.content)

# Run the async loop
asyncio.run(generate_pure_code())
>>>python
